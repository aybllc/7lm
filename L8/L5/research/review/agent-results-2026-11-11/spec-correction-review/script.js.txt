  LENSES,
  l => agent(COMMON + '\n\n' + l.prompt, { label: 'review:' + l.key, phase: 'Review', schema: FIND }),

    return agent(COMMON + `

YOUR TASK (label: verify-${l.key}): a reviewer with the lens "${l.key}" reported the findings below about commit 33c3878. For each one, try to refute it. ${JSON.stringify(review.findings, null, 1)}`, { label: 'verify:' + l.key, phase: 'Verify', schema: VERD })
