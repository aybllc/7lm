` + JSON.stringify(reads, null, 1), { label: 'synthesize', phase: 'Synthesize', schema: SYNTH_SCHEMA })

phase('Verify')
const VERDICT = { type: 'object', required: ['refuted', 'confidence', 'reason'], properties: { refuted: { type: 'boolean' }, confidence: { type: 'string', enum: ['high', 'medium', 'low'] }, reason: { type: 'string' }, correction: { type: 'string', description: 'if refuted or imprecise, the corrected statement with evidence' } } }
const LENSES = [
  'textual accuracy: re-open the cited files or commits and check every quoted phrase, number, date and path in the claim; refute if any is wrong or unsupported',
  'omission and framing: check whether the claim leaves out a fact in the same source or a neighbouring file that changes its meaning (another version, a later commit, a contrary status label); refute if so',
  'chronology and location: check with git (log, show, ls-tree, branch -r --contains) that the dates, the branch or history location, and the "last changed" statements are right; refute if any is off',
]
const verified = await pipeline(synth.key_claims, c => parallel(LENSES.map((lens, i) => () =>
  agent(RULES + `\n\nTASK (verify, lens ${i + 1}): Try to REFUTE this claim. Default to refuted=true if you cannot confirm it from the files. Lens: ${lens}.\n\nCLAIM ${c.id}: ${c.claim}\nEVIDENCE OFFERED: ${c.evidence}`, { label: `verify:${c.id}:${i + 1}`, phase: 'Verify', schema: VERDICT })))
  .then(vs => ({ ...c, verdicts: vs.filter(Boolean), survives: vs.filter(Boolean).filter(v => !v.refuted).length >= 2 })))
const refutedClaims = verified.filter(Boolean).filter(c => !c.survives)
log(`claims: ${verified.filter(Boolean).length}, surviving: ${verified.filter(Boolean).length - refutedClaims.length}, refuted or unconfirmed: ${refutedClaims.length}`)

phase('Critique')
const CRITIC = { type: 'object', required: ['gaps'], properties: { gaps: { type: 'array', items: { type: 'object', required: ['gap', 'finding'], properties: { gap: { type: 'string' }, finding: { type: 'string', description: 'what you found when you went and looked, with evidence' } } } } } }

Ask what is missing: a source not read (search the repository and the branch for any manuscript, book, chapter, draft, prospectus, sourcebook, spine, TOC or block file the report does not mention: use git ls-files, git ls-tree -r ${BRANCH}, git grep -il for manuscript|prospectus|chapter|spine|sourcebook|TOC), a thread not covered, a date or location the report leaves vague, a claim the report makes without evidence, or a decision the records leave to the owner that the report omits. For each gap, go and look, and report what you found with evidence.\n\nDRAFT REPORT:\n` + synth.report, { label: 'critic', phase: 'Critique', schema: CRITIC })

phase('Final')
const FINAL = { type: 'object', required: ['report'], properties: { report: { type: 'string' } } }
const fin = await agent(`You are finishing a status report for a repository owner who asked: "Read the auditonomous manuscript dir and find out where the manuscript development got to." Rewrite the draft below into the final report, applying the verification verdicts (drop or correct any refuted or unconfirmed claim, using the corrections given) and closing the gaps the critic found (fold their findings in). Keep it plain English, concrete and dated, with the Markdown timeline table, 900-1700 words. Lead with the answer in three or four sentences. Keep paths in backticks and keep them sparse in prose. No em dashes. Do not address the reader as "the owner" in the second person; write "you" where needed. End with the decisions the records leave to you, as a short list.

DRAFT:
${synth.report}

VERIFICATION VERDICTS:
${JSON.stringify(verified.filter(Boolean).map(c => ({ id: c.id, claim: c.claim, survives: c.survives, verdicts: c.verdicts })), null, 1)}

CRITIC'S GAPS:
${JSON.stringify(critic.gaps, null, 1)}`, { label: 'final', phase: 'Final', schema: FINAL })

return { report: fin.report, claims_total: verified.filter(Boolean).length, claims_refuted: refutedClaims.map(c => ({ id: c.id, claim: c.claim, corrections: c.verdicts.map(v => v.correction).filter(Boolean) })), gaps: critic.gaps, readers: reads.map(r => ({ key: r.key, facts: (r.facts || []).length, open_questions: r.open_questions })) }
