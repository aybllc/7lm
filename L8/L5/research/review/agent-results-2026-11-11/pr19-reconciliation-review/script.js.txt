  LENSES,
  l => agent(COMMON + '\n\n' + l.prompt, { label: 'review:' + l.key, phase: 'Review', schema: FIND }),

    if (!review) return { lens: l.key, review: null, verdicts: null }
    if (!review.findings.length) return { lens: l.key, review, verdicts: [] }
    return agent(COMMON + `

YOUR TASK (label: verify-${l.key}): a reviewer with the lens "${l.key}" reported the findings below about merge commit 38e2ec3. For each one, try to refute it. Re-derive the evidence yourself from the files, under the same read-only rules; do not take the reviewer's evidence on trust. FINDINGS:
${JSON.stringify(review.findings, null, 1)}`, { label: 'verify:' + l.key, phase: 'Verify', schema: VERD })
