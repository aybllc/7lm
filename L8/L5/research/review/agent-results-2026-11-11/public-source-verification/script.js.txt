const VERIFY_SCHEMA = {
  type: 'object',
  properties: {
    verdicts: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          real: { type: 'boolean' },
          reasoning: { type: 'string' },
          corrected: { type: 'string', description: 'the precise, corrected statement of the problem, with the source words' },
        },
        required: ['id', 'real', 'reasoning'],
      },
    },
  },
  required: ['verdicts'],
}

const results = await pipeline(
  CHECKS,
  c => agent(RULES + '\n' + c.prompt + `\n\nYour label is "${c.key}". Return the structured object. Give every problem a short id (${c.key}-1, ${c.key}-2, ...).`, { label: `check:${c.key}`, phase: 'Check', schema: CHECK_SCHEMA }),
  (r, c) => {
    if (!r) return null
    if (!r.problems || r.problems.length === 0) return { key: c.key, check: r, verdicts: [] }
    return agent(RULES + `
YOU ARE A SKEPTIC for the check "${c.key}". A checker compared repository text with an official source and reported the problems below. PDF text extraction produces false alarms (column breaks, hyphenation, ligatures, footnote markers, running headers inside a passage, bracketed assignment placeholders split across lines). For EACH problem, re-establish the source text YOURSELF: you may reuse the checker's downloaded file only after confirming it is the official document (re-download it from the official host into your own scratch directory and compare sha256), then extract the passage again, by a different method where you can (for a PDF, try pdftotext without -layout as well as with it), and compare with the repository text read from the git object. A problem is REAL only if the WORDS, the attribution, the count or the recorded identity are in fact wrong. Return one verdict per problem id, and for each real one give the precise corrected statement with the source's words and their place.
The checker's sources (with the paths where it saved them): ${JSON.stringify(r.sources)}
PROBLEMS (JSON): ${JSON.stringify(r.problems)}`, { label: `verify:${c.key}`, phase: 'Verify', schema: VERIFY_SCHEMA, effort: 'high' })
      .then(v => ({ key: c.key, check: r, verdicts: v ? v.verdicts : null }))
  }
)

const out = { checks: [], confirmed_problems: [], rejected_problems: [], unverified_problems: [], failed_checks: [] }
CHECKS.forEach((c, i) => {
  const r = results[i]
  if (!r) { out.failed_checks.push(c.key); return }
  const k = r.check
  out.checks.push({ key: c.key, items_total: k.items_total, confirmed: k.confirmed, problems_reported: (k.problems || []).length, sources: k.sources, could_not_check: k.could_not_check || [], summary: k.summary })
  const vmap = {}
  if (r.verdicts) for (const v of r.verdicts) vmap[v.id] = v
  for (const p of (k.problems || [])) {
    const v = vmap[p.id]
    if (!r.verdicts || !v) { out.unverified_problems.push(Object.assign({ check: c.key }, p)); continue }
    if (v.real) out.confirmed_problems.push(Object.assign({ check: c.key }, p, { corrected: v.corrected || '', skeptic: v.reasoning }))
    else out.rejected_problems.push({ check: c.key, id: p.id, repo_locator: p.repo_locator, why_not: v.reasoning })
  }
})
log(`checks returned ${out.checks.length}/${CHECKS.length}; items tested ${out.checks.reduce((n, c) => n + (c.items_total || 0), 0)}, confirmed ${out.checks.reduce((n, c) => n + (c.confirmed || 0), 0)}; problems confirmed ${out.confirmed_problems.length}, rejected ${out.rejected_problems.length}, unverified ${out.unverified_problems.length}`)
return out
