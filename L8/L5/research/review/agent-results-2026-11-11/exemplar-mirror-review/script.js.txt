const FINDINGS_SCHEMA = {
  type: 'object',
  properties: {
    findings: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          severity: { type: 'string', enum: ['blocker', 'major', 'minor'] },
          file: { type: 'string' },
          line: { type: 'string' },
          statement: { type: 'string', description: 'what is wrong, concretely' },
          evidence: { type: 'string', description: 'locators and quotes that show it' },
          suggested_fix: { type: 'string', description: 'the exact replacement text or operation' },
        },
        required: ['id', 'severity', 'file', 'statement', 'evidence'],
      },
    },
    checked: { type: 'array', items: { type: 'string' } },
  },
  required: ['findings', 'checked'],
}

phase('Review')
const lenses = ['complete', 'harm', 'record']
const jobs = []
for (const r of REPOS) for (const l of lenses) jobs.push({ r, l })
const reviews = await parallel(jobs.map(j => () =>
  agent(RULES + '\n' + lensPrompt(j.r, j.l), { label: `review:${j.r.key}:${j.l}`, phase: 'Review', schema: FINDINGS_SCHEMA })
    .then(x => (x ? { repo: j.r.key, lens: j.l, findings: x.findings || [], checked: x.checked || [] } : null))
))
const got = reviews.filter(Boolean)
const all = []
for (const g of got) for (const f of g.findings) all.push(Object.assign({ repo: g.repo, lens: g.lens }, f))
log(`reviews returned: ${got.length}/${jobs.length}; findings: ${all.length}`)

const VERDICT_SCHEMA = {
  type: 'object',
  properties: {
    real: { type: 'boolean' },
    reasoning: { type: 'string' },
    corrected_statement: { type: 'string' },
    exact_fix: { type: 'string' },
  },
  required: ['real', 'reasoning'],
}

phase('Verify')
const judged = await parallel(all.map(f => () => {
  const r = REPOS.find(x => x.key === f.repo)
  return agent(RULES + `
YOU ARE A SKEPTIC. A reviewer reported the finding below against ${r.key} (${r.path}, commit ${r.commit}, parent ${r.base}). Decide whether it is REAL: a true defect of the commit measured against the intended content stated above, shown by the git objects. Check the evidence yourself. It is NOT real if the thing is intended, deliberately not done, a matter of taste, already true of the parent and outside the pass's scope, or simply wrong. If it is real, give the exact fix (the replacement text or the operation), as small as possible.
FINDING ${f.id} [${f.severity}] in ${f.file}${f.line ? ':' + f.line : ''}
STATEMENT: ${f.statement}
EVIDENCE: ${f.evidence}
SUGGESTED FIX: ${f.suggested_fix || '(none given)'}`, { label: `verify:${f.repo}:${f.id}`, phase: 'Verify', schema: VERDICT_SCHEMA, effort: 'high' })
    .then(v => Object.assign({}, f, { verdict: v }))
}))

const confirmed = judged.filter(Boolean).filter(j => j.verdict && j.verdict.real)
const rejected = judged.filter(Boolean).filter(j => j.verdict && !j.verdict.real)
log(`confirmed: ${confirmed.length}; rejected: ${rejected.length}`)
return {
  confirmed,
  rejected: rejected.map(j => ({ repo: j.repo, id: j.id, statement: j.statement, why_not: j.verdict.reasoning })),
  checked: got.map(g => ({ repo: g.repo, lens: g.lens, checked: g.checked })),
  reviews_returned: got.length,
}
