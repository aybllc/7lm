export const meta = {
  name: 'exemplar-sheet-audit',
  description: 'Read-only audit of every live sheet of one exemplar repository against the disk, the specification and the other sheets; each finding tested by a skeptic',
  phases: [
    { title: 'Audit', detail: 'one auditor per subtree batch of sheets' },
    { title: 'Verify', detail: 'one skeptic per batch tests every finding' },
  ],
}

const R = args
const SCRATCH = '/tmp/claude-0/-home-user/bfcfd013-a2da-5019-961e-7f46116f08be/scratchpad/wf-audit-' + R.key
const SPEC = 'L8/L4/work/writing/7lm-harmonized-architecture-specification.md'

const RULES = `You are one agent in a READ-ONLY audit. Nothing may be changed.

THE REPOSITORY: ${R.key}, local clone ${R.path}, PINNED COMMIT ${R.commit} (read this commit and no other state). It is an "exemplar" of the Seven-Layer Model (7LM). The model's scaffold is /home/user/7lm (model of record = origin/main). The controlling text is the specification: /home/user/7lm origin/main:${SPEC} (about 1,000 long lines).

THE LAYOUT. The root holds README.md, L7/, L8/, .github/, .gitignore and nothing else. L7/ is the desk: provenance (prior-art, sources, lineage, authority); governance (copyright, licensing, patents, permissions, access, controlled); peering ({private,public}/{ingress,egress}/peer-N/object-M); allications (publications, applications, communications, specifications, notifications, participations). L8/ is the interior: L0 semantics and foundations (epistemic, ontology, pedagogy, semantics/definitions, universality); L1 state-space; L2 formal-mathematics (relations, operators, proofs); L3 engineering-mathematics (approximation, uncertainty, tolerances); L4 work (computation, experiments, writing); L5 research (lanes, review, findings) and institution (peers, undergraduates); L6 history and retired. Every directory carries a "sheet", a file named 0.md, whose first line is "# <path>/". A sheet states what its directory is, why it exists, what it holds and does not hold; many carry a "Holdings" section that lists what is actually filed there. Retired sets under L8/L6/retired/<set>/ are kept verbatim and are NOT audited below their root. Records under L8/L6/history/ say what happened and keep the names that were true when written; they are NOT audited for current names.

WHAT JUST CHANGED. Commit ${R.commit} (2026-11-11) mirrored the scaffold's L7 restructure: L7/publications/ moved to L7/allications/publications/, five sibling positions and a container sheet were added, L7/governance/restrictions/ moved to L7/governance/controlled/, and the words in live sheets were made to match. Its record is section ${R.sec} of L8/L6/history/harmonization-2026-09-05.md. That restructure is intended; audit what the sheets now say.

HARD RULES
1. READ-ONLY. Never run git checkout, switch, restore, reset, merge, rebase, commit, stash, push, fetch, pull, worktree, clean, gc, or anything that writes inside ${R.path} or /home/user/7lm.
2. NEVER read a file from a working tree. Read only git objects at the pinned commit: \`git -C ${R.path} show ${R.commit}:<path>\`, \`git -C ${R.path} ls-tree -r --name-only ${R.commit} [path]\`, \`git -C ${R.path} ls-tree -r -l ${R.commit} <path>\` (sizes), \`git -C ${R.path} grep -n -F '<text>' ${R.commit} -- <paths>\`. For the specification: \`git -C /home/user/7lm show origin/main:${SPEC}\` and grep on that ref. Quote paths that contain spaces, '#' or parentheses.
3. Scratch files only under ${SCRATCH}/<your label>/ (create it). You may copy an object there to count rows or lines, or to extract text from an Office file with python3 -I and zipfile.
4. Every finding carries the sheet's path and line, the sheet's exact words (quoted), the evidence of what is actually so (a command result, a locator, a quote), and the exact replacement text that would make the sheet true.
5. Do not guess. If you cannot determine the truth of a statement, do not report it as a finding; list it under "could_not_check".

WHAT COUNTS AS A FINDING (report only these kinds)
- WRONG-FACT: the sheet states something about this repository that the tree at the commit shows to be false: a file or folder it names that is not there; a count of files, rows, lines, items, definitions, sources or bytes that is off; "nothing is filed" where something is; "holds X" where X is elsewhere.
- DEAD-PATH: a path the sheet names (in this repository) that does not exist at the commit, unless the sheet presents it as a former path or as a path in another repository.
- MISQUOTE: text the sheet presents as the specification's (in quotation marks attributed to the specification, or after "The specification's prose cell:" or similar) that is not verbatim in the current specification; or a cell quoted for this position that is in fact the specification's cell for a different position.
- CONTRADICTION: two sheets, or a sheet and the README, state the same fact about this repository differently. Give both locators.
- STALE: a statement that was true at an earlier state and is untrue now (for example "this pass", "at HEAD", "pending", a pointer to something since moved).
- UNLISTED (observation): a file or sub-folder that exists in the directory and that the sheet's account of what is held does not mention at all.
- MISFILED (observation): a file in the directory that the sheet's own "Does not hold" list, or the specification's row for the position, says belongs at another position. Say where it would belong and why; this is for the owner to decide, not a defect to fix.

WHAT IS NOT A FINDING
- Wording, style, length, or any difference from the scaffold's sheet at the same path: what a sheet says beyond its first line is the repository's own choice.
- The specification's Object and Function cells for L7, which still read "peering, publications, exposure, and exchange": they are quoted correctly.
- The word "restrictions" where it names a kind of condition and not the folder.
- Already known and recorded in section ${R.sec} of the record; do not report again: ${R.known}
- Anything inside L8/L6/history/ (other than its 0.md), below a retired set, or inside research documents, staging records and source texts: you read those only as evidence of what is held.
`

const FIND_SCHEMA = {
  type: 'object',
  properties: {
    sheets_audited: { type: 'array', items: { type: 'string' } },
    statements_checked: { type: 'integer', description: 'how many checkable statements you tested across the batch' },
    findings: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          kind: { type: 'string', enum: ['WRONG-FACT', 'DEAD-PATH', 'MISQUOTE', 'CONTRADICTION', 'STALE', 'UNLISTED', 'MISFILED'] },
          sheet: { type: 'string' },
          line: { type: 'string' },
          sheet_says: { type: 'string' },
          truth: { type: 'string' },
          evidence: { type: 'string' },
          exact_fix: { type: 'string' },
        },
        required: ['id', 'kind', 'sheet', 'sheet_says', 'truth', 'evidence'],
      },
    },
    clean_sheets: { type: 'array', items: { type: 'string' } },
    could_not_check: { type: 'array', items: { type: 'string' } },
  },
  required: ['sheets_audited', 'findings'],
}

const VERIFY_SCHEMA = {
  type: 'object',
  properties: {
    verdicts: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          real: { type: 'boolean' },
          reasoning: { type: 'string' },
          corrected_truth: { type: 'string' },
          exact_fix: { type: 'string' },
        },
        required: ['id', 'real', 'reasoning'],
      },
    },
  },
  required: ['verdicts'],
}

const findPrompt = b => RULES + `
YOUR BATCH (${b.key}): every sheet (file named 0.md) at or under these paths, at commit ${R.commit}: ${b.prefixes.join(' ; ')}${b.exclude && b.exclude.length ? '\nEXCLUDE anything under: ' + b.exclude.join(' ; ') : ''}${b.extra && b.extra.length ? '\nALSO AUDIT these files the same way: ' + b.extra.join(' ; ') : ''}
List them first (git ls-tree -r --name-only ${R.commit} <prefix> | grep -E '(^|/)0\\.md$'), then audit EVERY one; do not sample.

FOR EACH SHEET
1. Read it in full with line numbers.
2. List the directory it describes (files and sub-folders, with sizes).
3. Test every checkable statement:
   a. What it says is held: each named file exists; each count is right (count rows of a table or ledger, definitions, sources, files, lines, by reading the object); "nothing filed" or "empty" is true; descriptions of a held file match the file's own header (title, date, status line).
   b. Every path it names in this repository resolves at the commit.
   c. Every quotation attributed to the specification is verbatim in the current specification, and the prose cell and "why" cell it gives are the specification's cells for THIS position (find the row for the position in the specification's prose-and-rationale tables).
   d. Its tree drawings and tables of sub-positions match the sub-folders that exist.
   e. Its statements about other positions agree with those positions' own sheets (spot-check each cross-reference).
   f. Anything in the directory the sheet does not account for.
   g. Anything in the directory that the sheet's own "Does not hold" list places elsewhere.
4. Record findings. Record sheets with no finding under "clean_sheets".
Return the structured object. Give "statements_checked" as an honest count.`

const verifyPrompt = (b, f) => RULES + `
YOU ARE A SKEPTIC for batch ${b.key}. An auditor reported the findings below. For EACH one, decide independently whether it is REAL: check the sheet's words at the cited line, then establish the truth yourself from the git objects at commit ${R.commit} (and the current specification where the finding is a MISQUOTE). A finding is NOT real if the sheet does not say what is claimed, if the "truth" is wrong, if it is one of the things listed above as not a finding or as already known, or if it is a matter of wording. For an UNLISTED or MISFILED observation, "real" means the observation is factually accurate. If it is real, restate the truth precisely and give the exact replacement text, as small as possible, that would make the sheet true; if the auditor's fix is right, repeat it. Return one verdict per finding id; do not omit any.

FINDINGS (JSON):
${JSON.stringify(f.findings)}`

const results = await pipeline(
  R.batches,
  b => agent(findPrompt(b), { label: `audit:${R.key}:${b.key}`, phase: 'Audit', schema: FIND_SCHEMA }),
  (f, b) => {
    if (!f) return null
    if (!f.findings || f.findings.length === 0) return { batch: b.key, audit: f, verdicts: [] }
    return agent(verifyPrompt(b, f), { label: `verify:${R.key}:${b.key}`, phase: 'Verify', schema: VERIFY_SCHEMA, effort: 'high' })
      .then(v => ({ batch: b.key, audit: f, verdicts: v ? v.verdicts : null }))
  }
)

const out = { repo: R.key, commit: R.commit, batches: [], confirmed: [], rejected: [], unverified: [], sheets_audited: 0, statements_checked: 0, batches_failed: [] }
R.batches.forEach((b, i) => {
  const r = results[i]
  if (!r) { out.batches_failed.push(b.key); return }
  const a = r.audit
  out.sheets_audited += (a.sheets_audited || []).length
  out.statements_checked += a.statements_checked || 0
  out.batches.push({ batch: b.key, sheets: (a.sheets_audited || []).length, findings: (a.findings || []).length, clean_sheets: a.clean_sheets || [], could_not_check: a.could_not_check || [] })
  const vmap = {}
  if (r.verdicts) for (const v of r.verdicts) vmap[v.id] = v
  for (const f of (a.findings || [])) {
    const v = vmap[f.id]
    const row = Object.assign({ batch: b.key }, f)
    if (!r.verdicts || !v) { out.unverified.push(row); continue }
    if (v.real) out.confirmed.push(Object.assign(row, { truth: v.corrected_truth || f.truth, exact_fix: v.exact_fix || f.exact_fix, skeptic: v.reasoning }))
    else out.rejected.push({ batch: b.key, id: f.id, sheet: f.sheet, sheet_says: f.sheet_says, why_not: v.reasoning })
  }
})
log(`${R.key}: sheets audited ${out.sheets_audited}, statements checked ${out.statements_checked}, confirmed ${out.confirmed.length}, rejected ${out.rejected.length}, unverified ${out.unverified.length}, failed batches ${out.batches_failed.length}`)
return out
